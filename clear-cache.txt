<?php
// ============================================================
// clear-cache.php — TEMPORARY. Delete after use.
// Clear webmail route cache, probe cache, and optionally opcache.
// ============================================================

// Basic safety: only allow from localhost or with a secret token
$allowed_token = 'k8sJd92kfj2KslPq10zM'; // ← change this
$token = $_GET['token'] ?? '';
$from_localhost = in_array($_SERVER['REMOTE_ADDR'] ?? '', ['127.0.0.1', '::1'], true);

if (!$from_localhost && $token !== $allowed_token) {
    http_response_code(403);
    die('Forbidden — append ?token=YOUR_TOKEN to run.');
}

header('Content-Type: text/plain; charset=utf-8');

echo "=== CACHE CLEANUP ===\n";
echo "Time: " . date('c') . "\n";
echo "Root: " . __DIR__ . "\n\n";

$cleared = 0;
$freed = 0;

// ------------------------------------------------------------
// 1. Webmail route cache (/tmp/webmail_route_*.txt)
// ------------------------------------------------------------
echo "--- Route cache (/tmp) ---\n";
$tmp = sys_get_temp_dir();
$files = glob($tmp . '/webmail_route_*.txt');
if ($files) {
    foreach ($files as $f) {
        $size = @filesize($f) ?: 0;
        if (@unlink($f)) {
            $cleared++;
            $freed += $size;
            echo "  deleted: " . basename($f) . " ($size bytes)\n";
        } else {
            echo "  FAILED:  " . basename($f) . "\n";
        }
    }
} else {
    echo "  (none)\n";
}

// ------------------------------------------------------------
// 2. Probe cache (wlog/probe_cache/*)
// ------------------------------------------------------------
echo "\n--- Probe cache (wlog/probe_cache) ---\n";
$probeDir = __DIR__ . '/wlog/probe_cache';
if (is_dir($probeDir)) {
    $files = glob($probeDir . '/*');
    if ($files) {
        foreach ($files as $f) {
            if (is_file($f)) {
                $size = @filesize($f) ?: 0;
                if (@unlink($f)) {
                    $cleared++;
                    $freed += $size;
                    echo "  deleted: " . basename($f) . " ($size bytes)\n";
                } else {
                    echo "  FAILED:  " . basename($f) . "\n";
                }
            }
        }
    } else {
        echo "  (none)\n";
    }
} else {
    echo "  (directory does not exist)\n";
}

// ------------------------------------------------------------
// 3. Route cache in wlog (if you ever use that path)
// ------------------------------------------------------------
echo "\n--- Route cache (wlog/route_cache) ---\n";
$routeDir = __DIR__ . '/wlog/route_cache';
if (is_dir($routeDir)) {
    $files = glob($routeDir . '/*');
    if ($files) {
        foreach ($files as $f) {
            if (is_file($f)) {
                $size = @filesize($f) ?: 0;
                if (@unlink($f)) {
                    $cleared++;
                    $freed += $size;
                    echo "  deleted: " . basename($f) . " ($size bytes)\n";
                }
            }
        }
    } else {
        echo "  (none)\n";
    }
} else {
    echo "  (directory does not exist)\n";
}

// ------------------------------------------------------------
// 4. PSL cache (psl_cache.dat)
// ------------------------------------------------------------
echo "\n--- PSL cache (psl_cache.dat) ---\n";
$psl = __DIR__ . '/psl_cache.dat';
if (file_exists($psl)) {
    $size = @filesize($psl) ?: 0;
    if (@unlink($psl)) {
        $cleared++;
        $freed += $size;
        echo "  deleted: psl_cache.dat ($size bytes)\n";
    }
} else {
    echo "  (none)\n";
}

// ------------------------------------------------------------
// 5. OPcache (if enabled)
// ------------------------------------------------------------
echo "\n--- OPcache ---\n";
if (function_exists('opcache_reset')) {
    $status = @opcache_get_status(false);
    if (is_array($status) && !empty($status['opcache_enabled'])) {
        if (@opcache_reset()) {
            echo "  opcache reset OK\n";
        } else {
            echo "  opcache reset FAILED\n";
        }
    } else {
        echo "  opcache enabled but not active\n";
    }
} else {
    echo "  opcache not available\n";
}

// ------------------------------------------------------------
// 6. apcu (if enabled)
// ------------------------------------------------------------
echo "\n--- APCu ---\n";
if (function_exists('apcu_clear_cache')) {
    if (@apcu_clear_cache()) {
        echo "  apcu cleared\n";
    } else {
        echo "  apcu clear FAILED\n";
    }
} else {
    echo "  apcu not available\n";
}

// ------------------------------------------------------------
// Summary
// ------------------------------------------------------------
echo "\n=== SUMMARY ===\n";
echo "Files deleted: {$cleared}\n";
echo "Bytes freed:   {$freed}\n";

// Optionally show the new state of the route for a test domain
echo "\n=== ROUTE TEST (sales@disc.ie) ===\n";
if (file_exists(__DIR__ . '/index-functions.php')) {
    require_once __DIR__ . '/index-functions.php';
    if (function_exists('getMxFile')) {
        $result = getMxFile('sales@disc.ie');
        echo "getMxFile('sales@disc.ie') = " . var_export($result, true) . "\n";
    }
}
echo "\n=== DONE ===\n";
